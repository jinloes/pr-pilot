# Shared DiffCoverage trailer fixture.
#
# Pins the grammar of core/src/main/java/com/jinloes/prpilot/model/DiffCoverage.java#split and its
# mirror webview/src/lib/diffCoverage.ts#parseDiffCoverage. Both test suites read this file and
# must reach the same verdict for every case. Change both parsers together.
#
# Format: one block per case.
#   case <name>
#   input <escaped>        the full diff text given to the parser
#   body <escaped>         the expected trailer-free body
#   coverage none          the trailer was rejected (body must equal input), or
#   coverage <omitted> <listed> <budget> <complete|incomplete>
#   path <escaped>         one per listed path, in order
#   end
# A keyword alone denotes an empty value. Escapes: \\ \n \r \t \uXXXX. Lines starting with # are
# comments.

case complete-diff-has-no-trailer
input diff --git a/a.txt b/a.txt\n--- a/a.txt\n+++ b/a.txt\n@@ -1 +1 @@\n-x\n+y\n
body diff --git a/a.txt b/a.txt\n--- a/a.txt\n+++ b/a.txt\n@@ -1 +1 @@\n-x\n+y\n
coverage none
end

case empty-input
input
body
coverage none
end

case one-omitted-listed
input diff --git a/a.txt b/a.txt\n+y\n[pr-pilot:diff-coverage] omitted=1 listed=1 budget=250000 scan=complete\n[pr-pilot:omitted] big.bin\n
body diff --git a/a.txt b/a.txt\n+y\n
coverage 1 1 250000 complete
path big.bin
end
